import os
import sys
import glob
import subprocess
import multiprocessing
 
def run_cmd(cmd, description):
    print(f"\n [RUNNING] {description}")
    print(f"Command: {cmd}")
    try:
        subprocess.run(cmd, shell=True, check=True)
        print(f" [SUCCESS] {description} finished.")
    except subprocess.CalledProcessError as e:
        print(f" [ERROR] Command failed during: {description}")
        sys.exit(1)

def find_fastq_files(directory, sample_id):
   
    r1_pattern = os.path.join(directory, f"*{sample_id}*_R1_*.fastq.gz")
    r2_pattern = os.path.join(directory, f"*{sample_id}*_R2_*.fastq.gz")
    
    r1_matches = glob.glob(r1_pattern)
    r2_matches = glob.glob(r2_pattern)
    
   
    if not r1_matches or not r2_matches:
        r1_pattern = os.path.join(directory, f"*{sample_id}*_1.fastq.gz")
        r2_pattern = os.path.join(directory, f"*{sample_id}*_2.fastq.gz")
        r1_matches = glob.glob(r1_pattern)
        r2_matches = glob.glob(r2_pattern)
        
    if r1_matches and r2_matches:
        return sorted(r1_matches)[0], sorted(r2_matches)[0]
    return None, None

def main():
   
    MAX_THREADS = max(1, multiprocessing.cpu_count() - 1)
    SAFE_THREADS = 4
    MEMORY_PER_THREAD = "2G"
    
    
    RAW_DATA_DIR = "/mnt/c/Users/venka/Downloads/ChIP_sequencing data/110626_chip seq raw files"
    REF_DIR = "./reference"
    BAM_DIR = "./bam_files"
    BIGWIG_DIR = "./bigwig"
    PEAKS_DIR = "./macs3_peaks"
    TRIM_DIR = "./trimmed_reads"
    WINDOWS_DOWNLOADS = "/mnt/c/Users/venka/Downloads"
    
    
    for folder in [BAM_DIR, BIGWIG_DIR, PEAKS_DIR, TRIM_DIR]:
        os.makedirs(folder, exist_ok=True)
        
    INDEX_PREFIX = os.path.join(REF_DIR, "spombe_index")
    
    
    sample_ids = ["2UM", "2TM", "2UP", "2TP"]
    
    print(f"📋 Initializing sequential processing pipeline for {len(sample_ids)} targets...")
    
    for sample_id in sample_ids:
        print(f"\n=======================================================")
        print(f" LOCATING & PROCESSING CHANNELS FOR: {sample_id}")
        print(f"=======================================================")
        
        
        r1_path, r2_path = find_fastq_files(RAW_DATA_DIR, sample_id)
        
        if not r1_path or not r2_path:
            print(f" [SKIP] Could not resolve unique file paths for sample ID: '{sample_id}' inside target directory.")
            continue
            
        print(f"Resolved R1 file: {os.path.basename(r1_path)}")
        print(f"Resolved R2 file: {os.path.basename(r2_path)}")
        
        
        name = f"{sample_id}_sample"
        
        trimmed_r1 = os.path.join(TRIM_DIR, f"{name}_val_1.fq.gz")
        trimmed_r2 = os.path.join(TRIM_DIR, f"{name}_val_2.fq.gz")
        
        sam_output = os.path.join(BAM_DIR, f"{name}.sam")
        bam_raw = os.path.join(BAM_DIR, f"{name}_raw.bam")
        bam_fixmate = os.path.join(BAM_DIR, f"{name}_fixmate.bam")
        bam_sorted = os.path.join(BAM_DIR, f"{name}_sorted.bam")
        bam_dedup = os.path.join(BAM_DIR, f"{name}_dedup.bam")
        bigwig_output = os.path.join(BIGWIG_DIR, f"{name}_cpm.bw")
        
       
        trim_cmd = f"trim_galore --paired -o \"{TRIM_DIR}\" --basename \"{name}\" \"{r1_path}\" \"{r2_path}\""
        run_cmd(trim_cmd, f"[{sample_id}] Executing Trim Galore base-clip protocols")
        
        
        align_cmd = f"bowtie2 -p {MAX_THREADS} -x \"{INDEX_PREFIX}\" -1 \"{trimmed_r1}\" -2 \"{trimmed_r2}\" -S \"{sam_output}\""
        run_cmd(align_cmd, f"[{sample_id}] Mapping reads via Bowtie2 paired-end mode")
        
        
        sam_to_bam_cmd = f"samtools view -h -b -S \"{sam_output}\" > \"{bam_raw}\""
        run_cmd(sam_to_bam_cmd, f"[{sample_id}] Generating binary compressed BAM blocks")
        
        
        fixmate_cmd = f"samtools fixmate -m -@ {MAX_THREADS} \"{bam_raw}\" \"{bam_fixmate}\""
        run_cmd(fixmate_cmd, f"[{sample_id}] Integrating structural mate flags")
        
        
        sort_cmd = f"samtools sort -@ {SAFE_THREADS} -m {MEMORY_PER_THREAD} -o \"{bam_sorted}\" \"{bam_fixmate}\""
        run_cmd(sort_cmd, f"[{sample_id}] Sorting BAM blocks by coordinates")
        
        
        dedup_cmd = f"samtools markdup -r -@ {MAX_THREADS} \"{bam_sorted}\" \"{bam_dedup}\""
        run_cmd(dedup_cmd, f"[{sample_id}] Removing PCR duplicate amplification bias")
        
        
        index_cmd = f"samtools index -@ {MAX_THREADS} \"{bam_dedup}\""
        run_cmd(index_cmd, f"[{sample_id}] Generating coordinate indexing tracks")
        
        
        cpm_cmd = f"bamCoverage -b \"{bam_dedup}\" -o \"{bigwig_output}\" -p {MAX_THREADS} --normalizeUsing CPM --binSize 10"
        run_cmd(cpm_cmd, f"[{sample_id}] Calculating CPM-normalized track data profiles")
        
        
        macs3_cmd = f"macs3 callpeak -t \"{bam_dedup}\" -f BAMPE -g 1.21e7 -n \"{name}\" --outdir \"{PEAKS_DIR}\" -q 0.05"
        run_cmd(macs3_cmd, f"[{sample_id}] Executing MACS3 peak caller engine")
        
        macs_bed = os.path.join(PEAKS_DIR, f"{name}_peaks.narrowPeak")
        final_bed = os.path.join(PEAKS_DIR, f"{name}_peaks.bed")
        if os.path.exists(macs_bed):
            os.rename(macs_bed, final_bed)
            
        
        print(f"Exporting analytical packages to Windows Host Space for {sample_id}...")
        run_cmd(f"cp \"{bam_dedup}\" \"{WINDOWS_DOWNLOADS}/\"", f"[{sample_id}] Syncing BAM track")
        run_cmd(f"cp \"{bam_dedup}.bai\" \"{WINDOWS_DOWNLOADS}/\"", f"[{sample_id}] Syncing BAM index track")
        run_cmd(f"cp \"{bigwig_output}\" \"{WINDOWS_DOWNLOADS}/\"", f"[{sample_id}] Syncing BigWig track")
        if os.path.exists(final_bed):
            run_cmd(f"cp \"{final_bed}\" \"{WINDOWS_DOWNLOADS}/\"", f"[{sample_id}] Syncing BED peak file")
            
        
        print(f" Flushing heavy processing data cache for sample: {sample_id}...")
        files_to_clean = [sam_output, bam_raw, bam_fixmate, bam_sorted, trimmed_r1, trimmed_r2]
        for file_path in files_to_clean:
            if os.path.exists(file_path):
                os.remove(file_path)
                
        print(f"Target profile processing for {sample_id} completed successfully.")
        
    print("\n [BATCH PIPELINE COMPLETE] Your de-duplicated BAM files, BigWig tracks, and peak BED files are ready inside your Windows Downloads folder!")

if __name__ == "__main__":
    main()
